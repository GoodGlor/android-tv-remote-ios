# Xiaomi TV Remote (iOS) — Phase 1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the complete SwiftUI UI of the TV remote app per the locked mockups, with all control traffic going through a mocked `TVController` seam.

**Architecture:** A local Swift package `RemoteCore` holds all testable logic (key model, controller protocol, mock controller, persistence) and is exercised with `swift test` on macOS. The iOS app target (generated by XcodeGen from `project.yml`) holds only SwiftUI views and theming, and depends on `RemoteCore`.

**Tech Stack:** Swift 6 / SwiftUI / Observation framework, Swift Testing (`import Testing`), XcodeGen (dev-only), iOS 17+.

**Spec:** `docs/superpowers/specs/2026-08-19-tv-remote-design.md`

## Global Constraints

- iOS deployment target **17.0**; iPhone only (`TARGETED_DEVICE_FAMILY 1`), portrait only.
- No external **runtime** dependencies. XcodeGen is a dev tool only.
- Colors verbatim from spec: background `#0C0D11`, surface `#14151A`, control `#16171C`, elevated `#1D1E24`, text `#F2F2F5` / `#8E8E96` / `#6E6E78`, hairline white@6%, accent `#FF8B3D`, connected `#34C759`, power tint `#FF6B61` on `rgba(255,69,58,0.14)`.
- All touch targets ≥ 44 pt.
- App shortcut tiles are **text labels only** (NETFLIX, YOUTUBE, PRIME, MI TV) — no brand logos.
- Views are dumb: render state, forward intents to `TVController`. No networking anywhere in Phase 1.
- After adding/removing files under `App/`, re-run `xcodegen generate` before building.
- Every commit message ends with the trailer line: `Co-Authored-By: Claude Fable 5 <noreply@anthropic.com>`.
- App build check command (used in several tasks):
  `xcodebuild -project RemoteControl.xcodeproj -scheme RemoteControl -destination 'generic/platform=iOS Simulator' build 2>&1 | tail -3` → expected `** BUILD SUCCEEDED **`.

```swift
public struct DiscoveredDevice: Identifiable, Hashable, Sendable {
    public let name: String
    public let host: String
    public var id: String { host }

    public init(name: String, host: String) {
        self.name = name
        self.host = host
    }
}

public struct PairedDevice: Codable, Equatable, Sendable {
    public let name: String
    public let host: String

    public init(name: String, host: String) {
        self.name = name
        self.host = host
    }
}

public enum ConnectionState: Equatable, Sendable {
    case disconnected, connecting, connected
}

/// The seam between UI and transport. Phase 1 ships only `MockTVController`;
/// Phase 2 adds a real Android TV implementation behind this same protocol.
@MainActor
public protocol TVController: AnyObject {
    var connectionState: ConnectionState { get }
    func discover() async -> [DiscoveredDevice]
    func pair(device: DiscoveredDevice, code: String) async
    func connect(to device: PairedDevice)
    func sendKey(_ key: KeyCommand)
    func sendText(_ text: String)
}
```

```swift
import Observation

/// Phase-1 stand-in for the real transport: simulates discovery and pairing,
/// and records every command so tests and debugging can inspect them.
@MainActor
@Observable
public final class MockTVController: TVController {
    public private(set) var connectionState: ConnectionState = .disconnected
    public private(set) var sentKeys: [KeyCommand] = []
    public private(set) var sentText: String = ""

    private let delay: Duration

    public init(delay: Duration = .milliseconds(300)) {
        self.delay = delay
    }

    public func discover() async -> [DiscoveredDevice] {
        try? await Task.sleep(for: delay)
        return [
            DiscoveredDevice(name: "Xiaomi TV P1e 32", host: "192.168.31.24"),
            DiscoveredDevice(name: "Mi Box S", host: "192.168.31.47"),
        ]
    }

    public func pair(device: DiscoveredDevice, code: String) async {
        connectionState = .connecting
        try? await Task.sleep(for: delay)
        connectionState = .connected
    }

    public func connect(to device: PairedDevice) {
        connectionState = .connected
    }

    public func sendKey(_ key: KeyCommand) {
        guard connectionState == .connected else { return }
        sentKeys.append(key)
    }

    public func sendText(_ text: String) {
        guard connectionState == .connected else { return }
        sentText = text
    }
}
```

```swift
import Foundation

/// Persists the single paired TV in UserDefaults.
public struct DeviceStore {
    private static let key = "pairedDevice"
    private let defaults: UserDefaults

    public init(defaults: UserDefaults = .standard) {
        self.defaults = defaults
    }

    public var pairedDevice: PairedDevice? {
        defaults.data(forKey: Self.key)
            .flatMap { try? JSONDecoder().decode(PairedDevice.self, from: $0) }
    }

    public func save(_ device: PairedDevice) {
        defaults.set(try? JSONEncoder().encode(device), forKey: Self.key)
    }

    public func clear() {
        defaults.removeObject(forKey: Self.key)
    }
}
```

`RemoteCore/Sources/RemoteCore/Preferences.swift`:

```swift
import Foundation

/// User-facing app preferences.
public struct Preferences {
    private let defaults: UserDefaults

    public init(defaults: UserDefaults = .standard) {
        self.defaults = defaults
    }

    /// Defaults to true — haptics on out of the box.
    public var hapticsEnabled: Bool {
        get { defaults.object(forKey: "hapticsEnabled") as? Bool ?? true }
        nonmutating set { defaults.set(newValue, forKey: "hapticsEnabled") }
    }

    /// Defaults to false — D-pad is the default navigation mode per spec.
    public var lastModeIsTouchpad: Bool {
        get { defaults.bool(forKey: "lastModeIsTouchpad") }
        nonmutating set { defaults.set(newValue, forKey: "lastModeIsTouchpad") }
    }
}
```

```swift
            }
        }
        .padding(18)
```
